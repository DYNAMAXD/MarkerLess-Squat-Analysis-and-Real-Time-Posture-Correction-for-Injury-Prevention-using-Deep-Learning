# Dependencies for app.py (the orchestrator + Gradio UI).
# AlphaPose's and MotionBERT's own requirements.txt files are installed
# separately inside the Dockerfile, from their cloned repos.
#
# gradio/pydantic are pinned deliberately: newer Pydantic (>=2.11) changed
# how it emits JSON-schema `additionalProperties: true`, which crashes
# gradio_client's schema builder with
#   "TypeError: argument of type 'bool' is not iterable"
# on launch (a widely-reported gradio issue, not anything pipeline-specific).
# Keep these two pins together; don't let pip float pydantic up on its own.
gradio==4.44.1
pydantic<2.11
numpy<2.0        # MotionBERT/AlphaPose era code assumes NumPy 1.x
pandas>=2.0
opencv-python-headless>=4.8
huggingface_hub>=0.23
ultralytics>=8.3        # only needed if you use --use_yolov11 (AGPL-3.0 — check licensing)
